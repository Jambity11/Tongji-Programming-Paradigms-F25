template<typename M, typename I, typename R, typename U>
void increaseEnumerate(int s, int e,
                       const M& match,
                       const I& insert,
                       const R& remove,
                       const U& update) {
  for (int l = s, r = s; l <= e; ) {
    while (l == r || (r <= e && !match(l, r - 1))) insert(l, r++);
    if (match(l, r - 1)) update(l, r - 2);
    else update(l, r - 1);
    remove(l++, r);
  }
}

template<bool Heuristic = true, bool PathCompression = true>
struct disjoint_set {
  vector<int> fa_or_sz;
  disjoint_set(int n) : fa_or_sz(n + 1, -1) {}
  int find(int u) {
    if constexpr (PathCompression) {
      if (fa_or_sz[u] < 0) return u;
      return fa_or_sz[u] = find(fa_or_sz[u]);
    } else {
      for ( ; fa_or_sz[u] >= 0; u = fa_or_sz[u]);
      return u;
    }
  }
  pair<int, int> unite (int u, int v) {
    int from = find(u), to = find(v);
    if (from == to) return {u, -1};
    if constexpr (Heuristic) {
      if (fa_or_sz[from] < fa_or_sz[to]) swap(from, to);
    }
    fa_or_sz[to] += fa_or_sz[from];
    fa_or_sz[from] = to;
    return {from, to};
  }
  bool is_united(int u, int v) { return find(u) == find(v); }
  int size(int u) { return -fa_or_sz[find(u)]; }
  int fa(int u) {return fa_or_sz[u] < 0 ? u : fa_or_sz[u]; }
  vector<vector<int>> group(int begin = 0, int end = -1) {
    if (end == -1) end = fa_or_sz.size() - 1;
    vector<vector<int>> res(fa_or_sz.size());
    for (int i = begin; i <= end; i ++) {
      res[find(i)].push_back(i); 
    }
    res.erase(
      remove_if(res.begin(), res.end(), [](vector<int>& v){ return v.empty();}), 
      res.end()
    );
    return res;
  }
};
